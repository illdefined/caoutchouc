% Hyperlink colour
\hypersetup { colorlinks = true, allcolors = hyperlink }

\bool_lazy_and:nnTF { \cs_if_exist_p:N \IfExistskomafont } { \cs_if_exist_p:N \addtokomafont } {
	\cs_new:Nn \__caoutchouc_koma_colour:nn {
		\IfExistskomafont { #1 } {
			\addtokomafont { #1 } {
				\color_select:n { #2 }
			}
		} { }
	}

	\keyval_parse:NNn \use_none:n \__caoutchouc_koma_colour:nn {
		subject = disposition,
		titlehead = disposition,
		title = disposition,
		subtitle = disposition,

		part = disposition,
		chapterprefix = disposition,
		disposition = disposition,
		captionlabel = disposition,

		%pagenumber = pagination,
		%footnotelabel = pagination,

		partentrynumber = pagination,
		chapterentrypagenumber = pagination,
		sectionentrynumber = pagination,

		enumeratelabel = pagination,
		itemizelabel = pagination,
	}

	\tl_if_exist:NT \partformat {
		\tl_gset:Nn \partformat {
			\partname\nobreakspace
			\soufre_colour_text:nn { pagination } { \thepart\autodot }
		}
	}

	\tl_if_exist:NT \chapterformat {
		\tl_gset:Nn \chapterformat {
			\mbox {
				\chapappifchapterprefix { \nobreakspace }
				\soufre_colour_text:nn { pagination } { \thechapter\autodot }
				\IfUsePrefixLine { } { \enskip }
			}
		}
	}

	\tl_map_inline:nn {
		\sectionformat \subsectionformat \subsubsectionformat \paragraphformat \subparagraphformat
	} {
		\tl_if_exist:NT #1 {
			\soufre_colour_wrap:Nn #1 { pagination }
		}
	}
} {
	\soufre_colour_wrap:Nn \thepage { pagination }
	\hook_gput_code:nnn { cmd/pagenumbering/after } { . } {
		\soufre_colour_wrap:Nn \thepage { pagination }
	}

	% Colour various item numbers
	\tl_map_inline:nn {
		\thepart \thechapter \thesection \thesubsection \theparagraph \thesubparagraph
		\thefigure \thetable
		\thefootnote \thempfootnote
		\labelenumi \labelenumii \labelenumiii \labelenumiv
		\labelitemi \labelitemii \labelitemiii \labelitemiv
	} {
		\tl_if_exist:NT #1 {
			\soufre_colour_wrap:Nn #1 { pagination }
		}
	}

	% Hook appendix initialisation as it re‐defines the top‐level sectioning
	\hook_gput_code:nnn { cmd/appendix/after } { . } {
		\tl_map_inline:nn {
			\thechapter \thesection
		} {
			\tl_if_exist:NT #1 {
				\soufre_colour_wrap:Nn #1 { pagination }
				\tl_map_break:
			}
		}
	}
}
