\cs_new:Nn \caoutchouc_uri:n {
	\group_begin:

	\keys_set:nn { caoutchouc / uri } { #1 }

	\l__caoutchouc_uri_scheme_str :
	\str_if_empty:NTF \l__caoutchouc_uri_authority_str {
		\bool_lazy_all:nF {
			{ \str_if_empty_p:N \l__caoutchouc_uri_userinfo_str }
			{ \str_if_empty_p:N \l__caoutchouc_uri_host_str }
			{
				\int_compare_p:VNn \l__caoutchouc_uri_port_int < { 0 }
			}
		} {
			//
			\str_if_empty:NF \l__caoutchouc_uri_userinfo_str {
				\l__caoutchouc_uri_userinfo_str @
			}
			\l__caoutchouc_uri_host_str
			\int_compare:VNnT \l__caoutchouc_uri_port_str >= { 0  } {
				: \int_to_arabic:V \l__caoutchouc_uri_port_int
			}
		}
	} {
		// \l__caoutchouc_uri_authority_str
	}

	\str_if_empty:NTF \l__caoutchouc_uri_authority_str {
		% Path cannot begin with // if no authority is defined
		\bool_lazy_and:nnT {
			\int_compare_p:nNn {
				\seq_count:N \l__caoutchouc_uri_path_seq
			} > { 3 }
		} {
			\tl_if_empty:e {
				\seq_item:Nn \l__caoutchouc_uri_path_seq { 2 }
			}
		} {
			\msg_fatal:nnn { caoutchouc } { uri-authority-ambiguity } { #1 }
		}
	} {
		% Path must be empty or absolute if authority is defined
		\bool_lazy_or:nnT {
			\seq_if_empty_p:N \l__caoutchouc_uri_path_seq
		} {
			\tl_if_empty_p:e {
				\seq_item:Nn \l__caoutchouc_uri_path_seq { 2 }
			}
		} {
			\msg_fatal:nnn { caoutchouc } { uri-authority-absolute } { #1 }
		}
	}

	\seq_use:Nn \l__caoutchouc_uri_path_seq { / }

	\str_if_empty:NF \l__caoutchouc_uri_query_str {
		?
		\l__caoutchouc_uri_query_str
	}

	\str_if_empty:NF \l__caoutchouc_uri_fragment_str {
		\#
		\l__caoutchouc_uri_fragment_str
	}

	\group_end:
}
