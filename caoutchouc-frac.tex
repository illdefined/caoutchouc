\cs_generate_variant:Nn \str_case:nnF { VnF }

\cs_new_protected:Nn \caoutchouc_numr_subst:n {
	\prop_get:NnNTF \c__caoutchouc_numr_subst_prop { #1 } \l__caoutchouc_numr_subst {
		\str_case:VnF { \l__caoutchouc_numr_subst } {
			{ numr } {
				{
					\addfontfeature { VerticalPosition = Numerator }
					#1
				}
			}

			{ sups } {
				\raisebox { -0.2ex } {
					\addfontfeature { VerticalPosition = Superior }
					#1
				}
			}
		} {
			\l__caoutchouc_numr_subst
		}
	} {
		\raisebox { 0.8ex } {
			\addfontfeature { Scale = 0.5 }
			\bfseries
			#1
		}
	}
}
