	\prop_if_in:NnTF \g__caoutchouc_locale_prop { #1 - #2 / #3 } {
		\prop_item:Nn \g__caoutchouc_locale_prop { #1 - #2 / #3 }
	} {
		\prop_if_in:NnTF \g__caoutchouc_locale_prop { #1 / #3 } {
			\prop_item:Nn \g__caoutchouc_locale_prop { #1 / #3 }
		} {
			\msg_expandable_error:nnnn { caoutchouc } { locale-unknown } { #1 } { #2 }
		}
	}
}
